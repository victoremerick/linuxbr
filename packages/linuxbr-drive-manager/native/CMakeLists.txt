cmake_minimum_required(VERSION 3.20)
project(linuxbr_drives LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Core DBus Xml)
find_package(KF6Bookmarks REQUIRED)
find_package(KF6KIO REQUIRED)
add_executable(linuxbr-places-sync places.cpp)
target_link_libraries(linuxbr-places-sync Qt6::Core Qt6::Xml KF6::Bookmarks)
add_library(kio_linuxbr MODULE worker.cpp)
target_link_libraries(kio_linuxbr Qt6::Core Qt6::DBus KF6::KIOCore)
set_target_properties(kio_linuxbr PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/kf6/kio")
