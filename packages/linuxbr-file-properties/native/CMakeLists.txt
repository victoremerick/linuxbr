cmake_minimum_required(VERSION 3.20)
project(linuxbr_file_properties LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(ECM 6.0 REQUIRED NO_MODULE)
list(APPEND CMAKE_MODULE_PATH ${ECM_MODULE_PATH})
include(KDEInstallDirs6)
find_package(Qt6 REQUIRED COMPONENTS Core Widgets)
find_package(KF6CoreAddons REQUIRED)
find_package(KF6KIO REQUIRED)
find_package(KF6WidgetsAddons REQUIRED)
add_library(linuxbrfileproperties MODULE properties.cpp)
target_link_libraries(linuxbrfileproperties PRIVATE Qt6::Core Qt6::Widgets KF6::CoreAddons KF6::KIOWidgets KF6::WidgetsAddons)
set_target_properties(linuxbrfileproperties PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/kf6/propertiesdialog")
install(TARGETS linuxbrfileproperties DESTINATION ${KDE_INSTALL_PLUGINDIR}/kf6/propertiesdialog)
option(BUILD_TESTING "Build the native dialog integration check" OFF)
if(BUILD_TESTING)
    enable_testing()
    add_executable(properties-dialog-check dialog-check.cpp)
    target_link_libraries(properties-dialog-check PRIVATE Qt6::Core Qt6::Widgets KF6::KIOWidgets KF6::WidgetsAddons)
    add_test(NAME properties-dialog-check COMMAND properties-dialog-check)
    add_executable(properties-multiple-check multiple-check.cpp)
    target_link_libraries(properties-multiple-check PRIVATE Qt6::Core Qt6::Widgets KF6::KIOWidgets KF6::WidgetsAddons)
    add_test(NAME properties-multiple-check COMMAND properties-multiple-check)
    add_executable(properties-open-check properties-open-check.cpp)
    target_link_libraries(properties-open-check PRIVATE Qt6::Core Qt6::Widgets KF6::KIOWidgets KF6::WidgetsAddons)
    set_tests_properties(properties-dialog-check PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_PLUGIN_PATH=${CMAKE_BINARY_DIR};LINUXBR_PROPERTIES_HELPER=${CMAKE_CURRENT_SOURCE_DIR}/../files/usr/lib/linuxbr-file-properties/backend.py")
    set_tests_properties(properties-multiple-check PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_PLUGIN_PATH=${CMAKE_BINARY_DIR}")
endif()
